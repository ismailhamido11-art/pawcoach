```javascript
export function isUserPremium(user) {
  if (!user) return false;
  if (user.is_premium) return true;         // Paye
  if (user.trial_expires_at) {
    return new Date(user.trial_expires_at) > new Date(); // Trial actif

  return false;
}
```

```javascript
const FREE_SCAN_LIMIT = 3; // const en haut du fichier

const checkScanLimit = (u) => {
  if (isUserPremium(u)) return false;
  const weekStart = getWeekStart();
  const isSameWeek = u?.scans_week_start === weekStart;
  const count = isSameWeek ? (u?.scans_this_week || 0) : 0;

};

// Appels actuels :
```
